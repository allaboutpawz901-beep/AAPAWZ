     1→"use client"
     2→
     3→import { useState, useMemo, useEffect } from "react"
     4→import { Check, CaretLeft, CaretRight, PawPrint, Calendar, Clock, Scissors, Dog, User, Note, CheckCircle, CreditCard } from "@phosphor-icons/react"
     5→
     6→type Breed = { id: string; name: string; sizeCategory: string; coatType: string; akcGroup?: string }
     7→type Service = { id: string; name: string; price: string; description?: string }
     8→type Groomer = { id: string; name: string; role: string; bio?: string }
     9→
    10→const SIZES = ["SMALL (0-20 lbs)", "MEDIUM (20-50 lbs)", "LARGE (50-90 lbs)", "X-LARGE (90+ lbs)"]
    11→
    12→export function BookingWizard({
    13→  breeds, services, groomers,
    14→}: {
    15→  breeds: Breed[]
    16→  services: Service[]
    17→  groomers: Groomer[]
    18→}) {
    19→  const [bookingType, setBookingType] = useState<"BOOKING" | "CONSULTATION" | "">("")
    20→  const [step, setStep] = useState(0)
    21→  const [ownerName, setOwnerName] = useState("")
    22→  const [phone, setPhone] = useState("")
    23→  const [email, setEmail] = useState("")
    24→  const [address, setAddress] = useState("")
    25→  const [dogName, setDogName] = useState("")
    26→  const [breedId, setBreedId] = useState("")
    27→  const [size, setSize] = useState(SIZES[0])
    28→  const [date, setDate] = useState("")
    29→  const [time, setTime] = useState("")
    30→  const [slots, setSlots] = useState<string[]>([])
    31→  const [loadingSlots, setLoadingSlots] = useState(false)
    32→  const [closed, setClosed] = useState(false)
    33→  const [serviceId, setServiceId] = useState("")
    34→  const [consultationReason, setConsultationReason] = useState("")
    35→  const [notes, setNotes] = useState("")
    36→  const [groomerId, setGroomerId] = useState("")
    37→  const [submitting, setSubmitting] = useState(false)
    38→  const [redirecting, setRedirecting] = useState(false)
    39→  const [success, setSuccess] = useState<"booking" | "consultation" | null>(null)
    40→
    41→  const selectedBreed = breeds.find((b) => b.id === breedId)
    42→  const selectedService = services.find((s) => s.id === serviceId)
    43→  const selectedGroomer = groomers.find((g) => g.id === groomerId)
    44→
    45→  useEffect(() => {
    46→    if (selectedBreed) {
    47→      const map: Record<string, string> = { "Small": SIZES[0], "Medium": SIZES[1], "Large": SIZES[2], "X-Large": SIZES[3] }
    48→      if (map[selectedBreed.sizeCategory]) setSize(map[selectedBreed.sizeCategory])
    49→    }
    50→  }, [breedId])
    51→
    52→  useEffect(() => {
    53→    if (!date) return
    54→    setLoadingSlots(true); setClosed(false)
    55→    fetch(`/api/availability?date=${date}`).then((r) => r.json()).then((d) => { setSlots(d.times || []); setClosed(!!d.closed); setTime("") }).catch(() => setSlots([])).finally(() => setLoadingSlots(false))
    56→  }, [date])
    57→
    58→  // Check URL for success param (return from Stripe)
    59→  useEffect(() => {
    60→    const params = new URLSearchParams(window.location.search)
    61→    if (params.get("success") === "booking") setSuccess("booking")
    62→    if (params.get("success") === "consultation") setSuccess("consultation")
    63→  }, [])
    64→
    65→  const steps = bookingType === "CONSULTATION"
    66→    ? ["Your Name", "Contact", "Your Dog", "Date & Time", "Reason", "Notes", "Groomer", "Confirm"]
    67→    : ["Your Name", "Contact", "Your Dog", "Date & Time", "Service", "Notes", "Groomer", "Deposit"]
    68→
    69→  const canNext = useMemo(() => {
    70→    if (bookingType === "") return false
    71→    switch (step) {
    72→      case 0: return !!ownerName.trim()
    73→      case 1: return !!phone.trim() && !!email.trim()
    74→      case 2: return !!dogName.trim() && !!breedId
    75→      case 3: return !!date && !!time
    76→      case 4: return bookingType === "CONSULTATION" ? !!consultationReason.trim() : !!serviceId
    77→      case 5: return true
    78→      case 6: return true
    79→      case 7: return true
    80→      default: return false
    81→    }
    82→  }, [step, ownerName, phone, email, dogName, breedId, date, time, serviceId, consultationReason, bookingType])
    83→
    84→  const submit = async () => {
    85→    setSubmitting(true)
    86→    try {
    87→      const res = await fetch("/api/bookings/checkout", {
    88→        method: "POST",
    89→        headers: { "Content-Type": "application/json" },
    90→        body: JSON.stringify({
    91→          bookingType, ownerName, phone, email, address,
    92→          dogName, breedId, breedName: selectedBreed?.name || "", size,
    93→          date, time, serviceId, serviceName: selectedService?.name || "",
    94→          consultationReason, notes, groomerId, groomerName: selectedGroomer?.name || "",
    95→          servicePrice: selectedService?.price || "",
    96→        }),
    97→      })
    98→      const data = await res.json()
    99→      if (data.url) {
   100→        setRedirecting(true)
   101→        window.location.href = data.url
   102→      } else if (data.type === "consultation") {
   103→        setSuccess("consultation")
   104→      }
   105→    } catch { /* ignore */ }
   106→    finally { setSubmitting(false) }
   107→  }
   108→
   109→  if (success) {
   110→    return (
   111→      <div className="border border-gold/30 bg-card p-10 text-center">
   112→        <div className="mx-auto flex h-16 w-16 items-center justify-center rounded-full bg-gold-deep">
   113→          <Check size={32} weight="bold" className="text-cream" />
   114→        </div>
   115→        <h2 className="mt-4 font-display text-[28px] text-ink">
   116→          {success === "booking" ? "Booking Confirmed" : "Consultation Requested"}
   117→        </h2>
   118→        <p className="script mt-2 text-[24px]">From Pawz to PAWfection</p>
   119→        <p className="mt-4 text-[13px] leading-relaxed text-ink-soft">
   120→          {success === "booking"
   121→            ? <>Your $25 deposit has been received and your appointment is confirmed. A confirmation email is on its way to {email}.</>
   122→            : <>We received your consultation request. Our team will reach out personally to schedule your visit.</>}
   123→        </p>
   124→        <a href="/" className="btn-gold mt-6 inline-flex">RETURN HOME</a>
   125→      </div>
   126→    )
   127→  }
   128→
   129→  // Step 0: Choose booking type
   130→  if (bookingType === "") {
   131→    return (
   132→      <div className="text-center">
   133→        <h2 className="font-display text-[28px] text-ink">How can we help your pup today?</h2>
   134→        <p className="mt-2 text-[12.5px] text-ink-soft">Choose an option to get started.</p>
   135→        <div className="mt-8 grid gap-4 sm:grid-cols-2">
   136→          <button onClick={() => { setBookingType("BOOKING"); setStep(0) }} className="group border border-gold/35 bg-cream p-8 text-center transition-colors hover:border-gold-deep hover:bg-cream-deep">
   137→            <Scissors size={32} weight="fill" className="mx-auto text-gold-deep" />
   138→            <h3 className="mt-3 font-display text-[20px] text-ink">Book Appointment</h3>
   139→            <p className="mt-1 text-[11.5px] leading-relaxed text-ink-soft">Schedule a grooming service. A $25 deposit secures your appointment.</p>
   140→          </button>
   141→          <button onClick={() => { setBookingType("CONSULTATION"); setStep(0) }} className="group border border-gold/35 bg-cream p-8 text-center transition-colors hover:border-gold-deep hover:bg-cream-deep">
   142→            <PawPrint size={32} weight="fill" className="mx-auto text-gold-deep" />
   143→            <h3 className="mt-3 font-display text-[20px] text-ink">Free Consultation</h3>
   144→            <p className="mt-1 text-[11.5px] leading-relaxed text-ink-soft">Meet with our team to discuss your pup&apos;s needs. No charge, no obligation.</p>
   145→          </button>
   146→        </div>
   147→      </div>
   148→    )
   149→  }
   150→
   151→  const stepIcons = [User, User, Dog, Calendar, Scissors, Note, PawPrint, bookingType === "CONSULTATION" ? Check : CreditCard]
   152→
   153→  return (
   154→    <div className="space-y-6">
   155→      {/* Stepper */}
   156→      <div className="flex items-center justify-between border-b border-gold/25 pb-5">
   157→        {steps.map((s, i) => {
   158→          const isActive = step === i
   159→          const isDone = step > i
   160→          const Icon = stepIcons[i]
   161→          return (
   162→            <div key={s} className="flex flex-1 items-center">
   163→              <button onClick={() => { if (i < step) setStep(i) }} disabled={i > step} className={`flex flex-col items-center gap-1.5 ${i < step ? "cursor-pointer" : "cursor-default"}`}>
   164→                <span className={`flex h-9 w-9 items-center justify-center rounded-full border-2 text-[12px] font-bold transition-colors ${isActive ? "border-gold-deep bg-gold-deep text-cream" : isDone ? "border-gold-deep bg-gold-deep text-cream" : "border-gold/30 bg-cream text-gold-deep"}`}>
   165→                  {isDone ? <Check size={14} weight="bold" /> : <Icon size={14} weight="fill" />}
   166→                </span>
   167→                <span className={`text-[8px] font-bold tracking-[0.08em] ${isActive ? "text-gold-deep" : "text-ink-soft"}`}>{s.toUpperCase()}</span>
   168→              </button>
   169→              {i < steps.length - 1 && <span className={`mx-1 h-px flex-1 ${step > i ? "bg-gold-deep" : "bg-gold/25"}`} />}
   170→            </div>
   171→          )
   172→        })}
   173→      </div>
   174→
   175→      <div className="min-h-[300px]">
   176→        {/* Step 0: Name */}
   177→        {step === 0 && (
   178→          <div className="space-y-5">
   179→            <div>
   180→              <h2 className="font-display text-[24px] text-ink">What is your name?</h2>
   181→              <p className="mt-1 text-[12px] text-ink-soft">So we know who to welcome.</p>
   182→            </div>
   183→            <div>
   184→              <label className="mb-1.5 block text-[9px] font-bold tracking-[0.16em] text-gold-deep">FULL NAME</label>
   185→              <input value={ownerName} onChange={(e) => setOwnerName(e.target.value)} placeholder="Jane Smith" autoFocus className="w-full border border-gold/35 bg-cream px-3.5 py-3 text-[14px] text-ink placeholder:text-muted-foreground focus:outline-none focus:ring-1 focus:ring-gold-deep" />
   186→            </div>
   187→          </div>
   188→        )}
   189→
   190→        {/* Step 1: Phone + email + address */}
   191→        {step === 1 && (
   192→          <div className="space-y-5">
   193→            <div>
   194→              <h2 className="font-display text-[24px] text-ink">Contact details</h2>
   195→              <p className="mt-1 text-[12px] text-ink-soft">We&apos;ll send your confirmation here.</p>
   196→            </div>
   197→            <div className="grid gap-4 sm:grid-cols-2">
   198→              <div>
   199→                <label className="mb-1.5 block text-[9px] font-bold tracking-[0.16em] text-gold-deep">PHONE</label>
   200→                <input value={phone} onChange={(e) => setPhone(e.target.value)} type="tel" placeholder="(312) 555-0142" className="w-full border border-gold/35 bg-cream px-3.5 py-3 text-[14px] text-ink placeholder:text-muted-foreground focus:outline-none focus:ring-1 focus:ring-gold-deep" />
   201→              </div>
   202→              <div>
   203→                <label className="mb-1.5 block text-[9px] font-bold tracking-[0.16em] text-gold-deep">EMAIL</label>
   204→                <input value={email} onChange={(e) => setEmail(e.target.value)} type="email" placeholder="jane@email.com" className="w-full border border-gold/35 bg-cream px-3.5 py-3 text-[14px] text-ink placeholder:text-muted-foreground focus:outline-none focus:ring-1 focus:ring-gold-deep" />
   205→              </div>
   206→              <div className="sm:col-span-2">
   207→                <label className="mb-1.5 block text-[9px] font-bold tracking-[0.16em] text-gold-deep">ADDRESS</label>
   208→                <input value={address} onChange={(e) => setAddress(e.target.value)} placeholder="123 Maple Grove Ave, Riverbend, IL 60614" className="w-full border border-gold/35 bg-cream px-3.5 py-3 text-[14px] text-ink placeholder:text-muted-foreground focus:outline-none focus:ring-1 focus:ring-gold-deep" />
   209→              </div>
   210→            </div>
   211→          </div>
   212→        )}
   213→
   214→        {/* Step 2: Dog */}
   215→        {step === 2 && (
   216→          <div className="space-y-5">
   217→            <div>
   218→              <h2 className="font-display text-[24px] text-ink">Tell us about your pup</h2>
   219→              <p className="mt-1 text-[12px] text-ink-soft">Every dog is special. We tailor the experience to their breed.</p>
   220→            </div>
   221→            <div className="grid gap-4 sm:grid-cols-2">
   222→              <div>
   223→                <label className="mb-1.5 block text-[9px] font-bold tracking-[0.16em] text-gold-deep">DOG NAME</label>
   224→                <input value={dogName} onChange={(e) => setDogName(e.target.value)} placeholder="Cooper" className="w-full border border-gold/35 bg-cream px-3.5 py-3 text-[14px] text-ink placeholder:text-muted-foreground focus:outline-none focus:ring-1 focus:ring-gold-deep" />
   225→              </div>
   226→              <div>
   227→                <label className="mb-1.5 block text-[9px] font-bold tracking-[0.16em] text-gold-deep">BREED</label>
   228→                <BreedDropdown breeds={breeds} value={breedId} onChange={setBreedId} />
   229→              </div>
   230→              <div className="sm:col-span-2">
   231→                <label className="mb-1.5 block text-[9px] font-bold tracking-[0.16em] text-gold-deep">SIZE</label>
   232→                <div className="flex flex-wrap gap-3">
   233→                  {SIZES.map((sz) => (
   234→                    <button key={sz} type="button" onClick={() => setSize(sz)} className={`border px-4 py-2.5 text-[11px] font-bold tracking-[0.1em] transition-colors ${size === sz ? "border-gold-deep bg-gold-deep text-cream" : "border-gold/35 bg-cream text-ink-soft hover:border-gold-deep"}`}>{sz}</button>
   235→                  ))}
   236→                </div>
   237→              </div>
   238→            </div>
   239→          </div>
   240→        )}
   241→
   242→        {/* Step 3: Date + Time */}
   243→        {step === 3 && (
   244→          <div className="space-y-5">
   245→            <div>
   246→              <h2 className="font-display text-[24px] text-ink">When would you like to come in?</h2>
   247→              <p className="mt-1 text-[12px] text-ink-soft">Open Tue–Sat 9am–6pm, Sun 10am–4pm. Closed Mondays.</p>
   248→            </div>
   249→            <div className="grid gap-5 lg:grid-cols-2">
   250→              <CalendarGrid value={date} onChange={setDate} />
   251→              <div>
   252→                {date ? (
   253→                  <>
   254→                    <p className="mb-3 text-[11px] font-bold tracking-[0.12em] text-gold-deep">AVAILABLE TIMES — {new Date(date + "T00:00").toLocaleDateString("en-US", { weekday: "short", month: "short", day: "numeric" })}</p>
   255→                    {loadingSlots ? <p className="text-[13px] text-ink-soft">Loading…</p> : closed ? <p className="text-[13px] text-ink-soft">We&apos;re closed on Mondays. Please pick another date.</p> : slots.length === 0 ? <p className="text-[13px] text-ink-soft">All slots are taken. Please try another day.</p> : (
   256→                      <div className="grid grid-cols-2 gap-2 sm:grid-cols-3">
   257→                        {slots.map((t) => <button key={t} onClick={() => setTime(t)} className={`border px-3 py-2.5 text-[12px] font-semibold transition-colors ${time === t ? "border-gold-deep bg-gold-deep text-cream" : "border-gold/35 bg-cream text-ink hover:border-gold-deep hover:bg-cream-deep"}`}>{t}</button>)}
   258→                      </div>
   259→                    )}
   260→                  </>
   261→                ) : <p className="text-[13px] text-ink-soft">Pick a date on the calendar to see available times.</p>}
   262→              </div>
   263→            </div>
   264→          </div>
   265→        )}
   266→
   267→        {/* Step 4: Service (booking) or Reason (consultation) */}
   268→        {step === 4 && bookingType === "BOOKING" && (
   269→          <div className="space-y-5">
   270→            <div>
   271→              <h2 className="font-display text-[24px] text-ink">Choose your service</h2>
   272→              <p className="mt-1 text-[12px] text-ink-soft">Select what your pup needs today.</p>
   273→            </div>
   274→            <div className="space-y-3">
   275→              {services.map((s) => (
   276→                <button key={s.id} onClick={() => setServiceId(s.id)} className={`flex w-full items-center justify-between border px-5 py-4 text-left transition-colors ${serviceId === s.id ? "border-gold-deep bg-cream-deep" : "border-gold/35 bg-cream hover:border-gold-deep"}`}>
   277→                  <div><p className="text-[13px] font-bold text-ink">{s.name}</p>{s.description && <p className="mt-0.5 text-[11px] text-ink-soft">{s.description}</p>}</div>
   278→                  <span className="text-[14px] font-bold text-gold-deep">{s.price}</span>
   279→                </button>
   280→              ))}
   281→            </div>
   282→          </div>
   283→        )}
   284→        {step === 4 && bookingType === "CONSULTATION" && (
   285→          <div className="space-y-5">
   286→            <div>
   287→              <h2 className="font-display text-[24px] text-ink">What brings you in?</h2>
   288→              <p className="mt-1 text-[12px] text-ink-soft">Tell us what you&apos;d like to discuss.</p>
   289→            </div>
   290→            <div className="space-y-2">
   291→              {["First visit / new puppy", "Coat or matting concerns", "Special needs or anxiety", "Breed recommendation", "Pricing question", "Product recommendation", "Other"].map((r) => (
   292→                <button key={r} onClick={() => setConsultationReason(r)} className={`flex w-full items-center border px-5 py-3 text-left text-[13px] transition-colors ${consultationReason === r ? "border-gold-deep bg-cream-deep font-semibold text-gold-deep" : "border-gold/35 bg-cream text-ink hover:border-gold-deep"}`}>
   293→                  {r}
   294→                </button>
   295→              ))}
   296→            </div>
   297→          </div>
   298→        )}
   299→
   300→        {/* Step 5: Notes */}
   301→        {step === 5 && (
   302→          <div className="space-y-5">
   303→            <div>
   304→              <h2 className="font-display text-[24px] text-ink">Special notes</h2>
   305→              <p className="mt-1 text-[12px] text-ink-soft">Anything we should know about your pup? Temperament, sensitivities, matting, etc.</p>
   306→            </div>
   307→            <textarea value={notes} onChange={(e) => setNotes(e.target.value)} rows={6} placeholder="e.g. Cooper is nervous around loud noises. He may need breaks during the groom. Sensitive paws." className="w-full border border-gold/35 bg-cream px-3.5 py-3 text-[14px] text-ink placeholder:text-muted-foreground focus:outline-none focus:ring-1 focus:ring-gold-deep" />
   308→          </div>
   309→        )}
   310→
   311→        {/* Step 6: Groomer */}
   312→        {step === 6 && (
   313→          <div className="space-y-5">
   314→            <div>
   315→              <h2 className="font-display text-[24px] text-ink">Preferred groomer</h2>
   316→              <p className="mt-1 text-[12px] text-ink-soft">Have a favorite? Or let us assign the best fit.</p>
   317→            </div>
   318→            <div className="space-y-3">
   319→              <button onClick={() => setGroomerId("")} className={`flex w-full items-center gap-4 border px-5 py-4 text-left transition-colors ${groomerId === "" ? "border-gold-deep bg-cream-deep" : "border-gold/35 bg-cream hover:border-gold-deep"}`}>
   320→                <PawPrint size={24} weight="fill" className="text-gold-deep" />
   321→                <div><p className="text-[13px] font-bold text-ink">No preference</p><p className="text-[11px] text-ink-soft">Assign the best available groomer</p></div>
   322→              </button>
   323→              {groomers.map((g) => (
   324→                <button key={g.id} onClick={() => setGroomerId(g.id)} className={`flex w-full items-center gap-4 border px-5 py-4 text-left transition-colors ${groomerId === g.id ? "border-gold-deep bg-cream-deep" : "border-gold/35 bg-cream hover:border-gold-deep"}`}>
   325→                  <div className="flex h-10 w-10 items-center justify-center rounded-full bg-gold-deep/15 font-display text-[16px] font-bold text-gold-deep">{g.name.charAt(0)}</div>
   326→                  <div><p className="text-[13px] font-bold text-ink">{g.name}</p><p className="text-[11px] text-ink-soft">{g.role}{g.bio ? ` — ${g.bio}` : ""}</p></div>

   329→            </div>

   332→
   333→        {/* Step 7: Review + Checkout */}

   335→          <div className="space-y-5">
   336→            <div>

   339→            </div>
   340→            <div className="border border-gold/30 bg-cream-deep p-5">
   341→              <p className="text-[10px] font-bold tracking-[0.18em] text-gold-deep">BOOKING SUMMARY</p>

   379→        {step < 7 ? (

   382→          </button>

   385→    </div>

   389→function BreedDropdown({ breeds, value, onChange }: { breeds: Breed[]; value: string; onChange: (v: string) => void }) {

   418→function CalendarGrid({ value, onChange }: { value: string; onChange: (v: string) => void }) {
   419→  const today = new Date()
   420→  const [viewMonth, setViewMonth] = useState(new Date(today.getFullYear(), today.getMonth(), 1))
   421→  const days = useMemo(() => {
   422→    const first = new Date(viewMonth.getFullYear(), viewMonth.getMonth(), 1)
   423→    const last = new Date(viewMonth.getFullYear(), viewMonth.getMonth() + 1, 0)
   424→    const cells: (Date | null)[] = []
   425→    for (let i = 0; i < first.getDay(); i++) cells.push(null)
   426→    for (let d = 1; d <= last.getDate(); d++) cells.push(new Date(viewMonth.getFullYear(), viewMonth.getMonth(), d))
   427→    return cells
   428→  }, [viewMonth])
   429→  const fmt = (d: Date) => `${d.getFullYear()}-${String(d.getMonth() + 1).padStart(2, "0")}-${String(d.getDate()).padStart(2, "0")}`
   430→  const isPast = (d: Date) => { const t = new Date(); t.setHours(0, 0, 0, 0); return d < t }
   431→  const isMonday = (d: Date) => d.getDay() === 1
   432→  return (
   433→    <div className="border border-gold/30 bg-card p-4">
   434→      <div className="mb-3 flex items-center justify-between">
   435→        <button onClick={() => setViewMonth(new Date(viewMonth.getFullYear(), viewMonth.getMonth() - 1, 1))} className="flex h-7 w-7 items-center justify-center text-gold-deep hover:bg-cream-deep"><CaretLeft size={14} weight="bold" /></button>
   436→        <span className="font-display text-[16px] text-ink">{viewMonth.toLocaleDateString("en-US", { month: "long", year: "numeric" })}</span>
   437→        <button onClick={() => setViewMonth(new Date(viewMonth.getFullYear(), viewMonth.getMonth() + 1, 1))} className="flex h-7 w-7 items-center justify-center text-gold-deep hover:bg-cream-deep"><CaretRight size={14} weight="bold" /></button>
   438→      </div>
   439→      <div className="mb-1 grid grid-cols-7 gap-1">{["S","M","T","W","T","F","S"].map((d, i) => <div key={i} className="text-center text-[9px] font-bold text-ink-soft">{d}</div>)}</div>
   440→      <div className="grid grid-cols-7 gap-1">
   441→        {days.map((d, i) => {
   442→          if (!d) return <div key={i} />
   443→          const ds = fmt(d)
   444→          const disabled = isPast(d) || isMonday(d)
   445→          return <button key={i} disabled={disabled} onClick={() => onChange(ds)} className={`aspect-square text-[11px] font-semibold transition-colors ${ds === value ? "bg-gold-deep text-cream" : disabled ? "text-ink-soft/30 cursor-not-allowed" : "text-ink hover:bg-cream-deep"}`}>{d.getDate()}</button>
   446→        })}
   447→      </div>
   448→    </div>
   449→  )
   450→}
